# pgBouncer userlist — one entry per line in the format:
#   "username" "md5<hash>"
#
# Generate the hash with (replace PASSWORD and USERNAME):
#   docker exec postgres psql -U postgres -c \
#     "SELECT concat('md5', md5('PASSWORDUSERNAME'));"
#
# Or for scram-sha-256 (postgres 17 default), copy the stored hash directly:
#   docker exec postgres psql -U postgres -c \
#     "SELECT rolpassword FROM pg_authid WHERE rolname = 'USERNAME';"
#
